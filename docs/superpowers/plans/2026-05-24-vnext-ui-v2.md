# vNext UI v2 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace the single-page basic vNext UI with a two-screen system (command center at `/` + chat at `/chat`) that uses the new organic+lattice brand, glass aesthetic, familiar ChatGPT-style sidebar, and pulls live data from new services for narrative greeting, GPU stats, and memory activity.

**Architecture:** Two self-contained HTML files served as static templates by Flask (`command_center.html` + `chat.html`), zero front-end framework, vanilla JS reading from existing REST surface plus three new endpoints (`/api/system/gpu`, `/api/memory/activity`, narrative composed server-side). New service modules live under `soveryn/app/services/`. No build step. SSE streaming preserved via `fetch+ReadableStream+AbortController`. Existing 498 tests stay green.

**Tech Stack:** Python 3.11, Flask 3, pytest 9, vanilla HTML/CSS/JS (no React/Vue/HTMX), SQLite (existing lattice + conversation stores), `subprocess` for `nvidia-smi`.

**Reference spec:** `docs/superpowers/specs/2026-05-24-vnext-ui-v2-design.md`

**Out of scope** (deferred per spec): Phase 4 region naming in code, lattice visualization, Attic schema, Visual memory storage, F (lattice writes from chat), G (tools / approval queue), `"Make Aetheria feel like Aetheria"` (SOUL.md prefix), bumping default `chat_timeout` 60→120. Mobile responsive is a separate pass.

```python
CHAT_TEMPLATE = Path(__file__).parent.parent / "templates" / "chat.html"


def _serve_chat_html():
    if not CHAT_TEMPLATE.is_file():
        return jsonify({"error": {
            "code": "ui_unavailable",
            "message": f"Chat template missing at {CHAT_TEMPLATE}",
        }}), 500
    html = CHAT_TEMPLATE.read_text(encoding="utf-8")
    resp = make_response(html, 200)
    resp.headers["Content-Type"] = "text/html; charset=utf-8"
    resp.headers["X-SOVERYN-UI-Source"] = "vnext-native"
    return resp


@bp.get("/chat")
def chat_index():
```

```html
    const m = path.match(/^\/chat\/([^/]+)$/);
    if (m) return { sessionId: m[1], agent: null };
    const sp = new URLSearchParams(window.location.search);
    return { sessionId: null, agent: sp.get("agent") };
  }

  function pickInitialAgent(urlAgent) {
    if (urlAgent && ACTIVE_AGENTS.includes(urlAgent.toLowerCase())) {
      return urlAgent.toLowerCase();
    }
    const stored = localStorage.getItem("soveryn.agent");
    if (stored && ACTIVE_AGENTS.includes(stored)) return stored;
    return "aetheria";
  }

  // ─── DOM helpers ───────────────────────────────────────────────────────────
  const $ = (sel) => document.querySelector(sel);
  const $$ = (sel) => Array.from(document.querySelectorAll(sel));
  const elThread = () => $('[data-testid="thread"]');
  const elInput  = () => $('[data-testid="composer-input"]');
  const elSend   = () => $('[data-testid="send"]');
  const elStreamToggle = () => $('[data-testid="stream-toggle"]');
  const elHistory = () => $('[data-testid="history"]');
  const elAgentName = () => $('[data-agent-name]');
  const elAgentDisc = () => $('[data-agent-disc]');
  const elSessionMeta = () => $('[data-session-meta]');

  function setAgent(name) {
    state.agent = name;
    localStorage.setItem("soveryn.agent", name);
    $$('.agent-pill').forEach(p => p.setAttribute("aria-pressed", String(p.dataset.agent === name)));
    elAgentName().textContent = name;
    elAgentDisc().style.background = `linear-gradient(135deg, ${AGENT_COLORS[name] || "var(--gold)"}, var(--bronze))`;
  }

  function appendBubble(role, content, opts = {}) {
    const div = document.createElement("div");
    div.className = `bubble ${role}`;
    if (opts.placeholder) {
      const span = document.createElement("span");
      span.className = "thinking-placeholder";
      span.textContent = "thinking…";
      div.appendChild(span);
    } else {
      div.textContent = content;
    }
    elThread().appendChild(div);
    elThread().scrollTop = elThread().scrollHeight;
    return div;
  }

  // ─── Streaming (POST + ReadableStream) ─────────────────────────────────────
  async function sendStreaming(message) {
    const userBubble = appendBubble("user", message);
    const respBubble = appendBubble("assistant", "", { placeholder: true });
    let cleared = false;

    state.abort = new AbortController();
    try {
      const r = await fetch("/chat_stream", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ agent: state.agent, session_id: state.sessionId, message }),
        signal: state.abort.signal,
      });
      const reader = r.body.getReader();
      const decoder = new TextDecoder();
      let buf = "";

      while (true) {
        const { value, done } = await reader.read();
        if (done) break;
        buf += decoder.decode(value, { stream: true });

        // SSE parse: chunks split by \n\n, each line starts "data: "
        const parts = buf.split("\n\n");
        buf = parts.pop();
        for (const part of parts) {
          const line = part.split("\n").find(l => l.startsWith("data: "));
          if (!line) continue;
          const payload = JSON.parse(line.slice(6));
          if (payload.type === "token" && payload.content) {
            if (!cleared) { respBubble.textContent = ""; cleared = true; }
            respBubble.textContent += payload.content;
            elThread().scrollTop = elThread().scrollHeight;
          } else if (payload.type === "error") {
            respBubble.textContent = `error · ${payload.code}: ${payload.message}`;
            return;
          } else if (payload.type === "session") {
            if (payload.session_id && !state.sessionId) {
              state.sessionId = payload.session_id;
              window.history.replaceState({}, "", `/chat/${payload.session_id}`);
            }
          }
        }
      }
    } catch (e) {
      if (e.name === "AbortError") {
        if (!cleared) respBubble.textContent = "(stopped)";
      } else {
        respBubble.textContent = `error · ${e.message}`;
      }
    } finally {
      state.abort = null;
    }
  }

  // ─── Sync send ─────────────────────────────────────────────────────────────
  async function sendSync(message) {
    appendBubble("user", message);
    const respBubble = appendBubble("assistant", "", { placeholder: true });
    try {
      const r = await fetch("/chat", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ agent: state.agent, session_id: state.sessionId, message }),
      });
      const data = await r.json();
      if (data.error) {
        respBubble.textContent = `error · ${data.error.code}: ${data.error.message}`;
      } else {
        respBubble.textContent = data.content || "";
        if (data.session_id && !state.sessionId) {
          state.sessionId = data.session_id;
          window.history.replaceState({}, "", `/chat/${data.session_id}`);
        }
      }
    } catch (e) {
      respBubble.textContent = `error · ${e.message}`;
    }
  }

  async function loadHistory(sessionId) {
    try {
      const r = await fetch(`/sessions/${sessionId}/history`);
      if (!r.ok) return;
      const data = await r.json();
      elThread().innerHTML = "";
      for (const t of data.turns || []) {
        if (t.role === "user" || t.role === "assistant") appendBubble(t.role, t.content);
      }
      elSessionMeta().textContent = `${(data.turns || []).length} messages`;
    } catch { /* noop */ }
  }

  // ─── Events ────────────────────────────────────────────────────────────────
  function bindEvents() {
    $$('.agent-pill').forEach(p => {
      p.addEventListener("click", () => {
        setAgent(p.dataset.agent);
        state.sessionId = null;
        elThread().innerHTML = "";
        window.history.replaceState({}, "", "/chat");
      });
    });

    elSend().addEventListener("click", onSubmit);
    elInput().addEventListener("keydown", (e) => {
      if (e.key === "Enter" && !e.shiftKey) { e.preventDefault(); onSubmit(); }
    });
    elStreamToggle().addEventListener("click", () => {
      state.stream = !state.stream;
      localStorage.setItem("soveryn.stream", String(state.stream));
      elStreamToggle().textContent = state.stream ? "stream ▾" : "sync  ▾";
    });

    $('[data-testid="new-chat"]').addEventListener("click", () => {
      state.sessionId = null;
      elThread().innerHTML = "";
      window.history.replaceState({}, "", "/chat");
      elInput().focus();
    });
  }

  async function onSubmit() {
    const msg = elInput().value.trim();
    if (!msg) return;
    elInput().value = "";
    if (state.stream) await sendStreaming(msg);
    else await sendSync(msg);
  }

  // ─── Boot ──────────────────────────────────────────────────────────────────
  function boot() {
    const { sessionId, agent } = parseUrl();
    setAgent(pickInitialAgent(agent));
    state.sessionId = sessionId;
    elStreamToggle().textContent = state.stream ? "stream ▾" : "sync  ▾";
    bindEvents();
    if (sessionId) loadHistory(sessionId);
  }
  document.addEventListener("DOMContentLoaded", boot);
})();
</script>
```

```python
"""Compose a plain-English greeting from session counts + lattice writes.

No LLM call — pure templating over typed input. Used by the command center
greeting block. Inputs are passed by the route layer; this module knows
nothing about Flask or SQLite.
"""

from __future__ import annotations
from dataclasses import dataclass, field
from datetime import datetime


@dataclass(frozen=True)
class GreetingInputs:
    now: datetime
    recent_writes_by_agent: dict[str, int] = field(default_factory=dict)
    recent_session_count: int = 0


@dataclass(frozen=True)
class Greeting:
    heading: str
    body: str


def _time_of_day(hour: int) -> str:
    if hour < 5:  return "Late, Jon."
    if hour < 12: return "Morning, Jon."
    if hour < 17: return "Afternoon, Jon."
    if hour < 22: return "Evening, Jon."
    return "Late, Jon."


def _writes_phrase(writes_by_agent: dict[str, int]) -> str:
    """e.g. 'Aetheria wrote 3 notes. Vett wrote 1 note.' or '' if empty."""
    parts: list[str] = []
    for agent, n in writes_by_agent.items():
        if n <= 0:
            continue
        noun = "note" if n == 1 else "notes"
        parts.append(f"{agent.capitalize()} wrote {n} {noun}.")
    return " ".join(parts)


def _sessions_phrase(n: int) -> str:
    if n <= 0:
        return ""
    return f"{n} session{'s' if n != 1 else ''} active in the last day."


def compose_greeting(inputs: GreetingInputs) -> Greeting:
    heading = _time_of_day(inputs.now.hour)

    writes = _writes_phrase(inputs.recent_writes_by_agent)
    sessions = _sessions_phrase(inputs.recent_session_count)

    parts = [p for p in (writes, sessions) if p]
    if not parts:
        body = "It's been quiet since you were last here."
    else:
        body = " ".join(parts)

    return Greeting(heading=heading, body=body)
```

```python
"""Wrap `nvidia-smi --query-gpu` for the command center system panel.

Subprocess-cached for 5 seconds in-process to avoid hammering the GPU on
every page poll. Gracefully degrades when nvidia-smi is missing (CI, dev
laptops without CUDA).
"""

from __future__ import annotations
import subprocess
import time
from dataclasses import dataclass, field

_CACHE_TTL_SECONDS = 5.0
_NVIDIA_SMI_TIMEOUT_SECONDS = 4.0
_QUERY_FIELDS = "index,utilization.gpu,temperature.gpu,memory.used,memory.total"


@dataclass(frozen=True)
class GpuStat:
    index: int
    util_pct: int
    temp_c: int
    mem_used_mib: float
    mem_total_mib: int


@dataclass(frozen=True)
class GpuStatsResult:
    available: bool
    gpus: list[GpuStat] = field(default_factory=list)
    message: str = ""
    fetched_at: float = 0.0


_cache: GpuStatsResult | None = None


def _parse_nvidia_smi(raw: str) -> list[GpuStat]:
    out: list[GpuStat] = []
    for line in raw.strip().splitlines():
        if not line.strip():
            continue
        cols = [c.strip() for c in line.split(",")]
        if len(cols) < 5:
            continue
        try:
            out.append(GpuStat(
                index=int(cols[0]),
                util_pct=int(float(cols[1])),
                temp_c=int(float(cols[2])),
                mem_used_mib=float(cols[3]),
                mem_total_mib=int(float(cols[4])),
            ))
        except ValueError:
            continue
    return out


def _run_nvidia_smi() -> GpuStatsResult:
    try:
        proc = subprocess.run(
            ["nvidia-smi", f"--query-gpu={_QUERY_FIELDS}", "--format=csv,noheader,nounits"],
            capture_output=True, text=True, timeout=_NVIDIA_SMI_TIMEOUT_SECONDS,
        )
    except FileNotFoundError:
        return GpuStatsResult(available=False, message="nvidia-smi not available", fetched_at=time.time())
    except subprocess.TimeoutExpired:
        return GpuStatsResult(available=False, message="nvidia-smi timed out", fetched_at=time.time())

    if proc.returncode != 0:
        return GpuStatsResult(
            available=False,
            message=f"nvidia-smi exit {proc.returncode}: {proc.stderr.strip() or 'no stderr'}",
            fetched_at=time.time(),
        )
    return GpuStatsResult(
        available=True,
        gpus=_parse_nvidia_smi(proc.stdout),
        fetched_at=time.time(),
    )


def get_gpu_stats(*, _force_refresh: bool = False) -> GpuStatsResult:
    """Return current GPU stats, cached for 5 seconds."""
    global _cache
    now = time.time()
    if not _force_refresh and _cache is not None and (now - _cache.fetched_at) < _CACHE_TTL_SECONDS:
        return _cache
    _cache = _run_nvidia_smi()
    return _cache
```

```python
"""Lattice write activity — daily counts and totals for the command center.

Reads directly from LatticeStore's underlying sqlite connection. The query
groups by date(created_at) for the last N days. Per-agent breakdown is
included so the activity feed can color-code.
"""

from __future__ import annotations
from dataclasses import dataclass, field
from datetime import datetime, timedelta, timezone

from soveryn.memory.lattice import LatticeStore


@dataclass(frozen=True)
class DailyBucket:
    date: str                 # ISO date, e.g. "2026-05-24"
    count: int
    by_agent: dict[str, int] = field(default_factory=dict)


@dataclass(frozen=True)
class MemoryActivity:
    days: int
    buckets: list[DailyBucket]  # oldest-first


def daily_write_counts(store: LatticeStore, *, days: int = 14, now: datetime | None = None) -> MemoryActivity:
    now = now or datetime.now(timezone.utc)
    today = now.date()
    earliest = today - timedelta(days=days - 1)

    with store._conn() as conn:
        rows = conn.execute(
            "SELECT date(created_at) AS d, agent, COUNT(*) AS n "
            "FROM nodes "
            "WHERE date(created_at) >= ? "
            "GROUP BY date(created_at), agent",
            (earliest.isoformat(),),
        ).fetchall()

    by_date: dict[str, dict[str, int]] = {}
    for row in rows:
        d = row["d"]
        by_date.setdefault(d, {})[row["agent"]] = row["n"]

    buckets: list[DailyBucket] = []
    for offset in range(days):
        d = (earliest + timedelta(days=offset)).isoformat()
        per_agent = by_date.get(d, {})
        buckets.append(DailyBucket(date=d, count=sum(per_agent.values()), by_agent=per_agent))
    return MemoryActivity(days=days, buckets=buckets)


def total_node_count(store: LatticeStore) -> int:
    with store._conn() as conn:
        row = conn.execute("SELECT COUNT(*) AS n FROM nodes").fetchone()
        return int(row["n"]) if row else 0
```

```python
"""SOVERYN vNext — /api/memory/* read-only memory stats."""

from __future__ import annotations
from dataclasses import asdict

from flask import Blueprint, current_app, jsonify, request

from soveryn.app.services.memory_activity import daily_write_counts, total_node_count
from soveryn.memory.lattice import LatticeStore

bp = Blueprint("api_memory", __name__)

_MAX_DAYS = 90
_DEFAULT_DAYS = 14


def _err(code: str, message: str, status: int):
    return jsonify({"error": {"code": code, "message": message}}), status


def _get_lattice_store() -> LatticeStore:
    """Resolve LatticeStore from app extensions, opening one if not cached."""
    state = current_app.extensions["soveryn"]
    store = state.get("lattice_store")
    if store is None:
        store = LatticeStore(state["env"].lattice_db)
        state["lattice_store"] = store
    return store


@bp.get("/api/memory/activity")
def api_memory_activity():
    raw = request.args.get("days", str(_DEFAULT_DAYS))
    try:
        days = int(raw)
    except ValueError:
        return _err("invalid_message", f"days must be an integer, got {raw!r}", 400)
    if days < 1:
        return _err("invalid_message", "days must be >= 1", 400)
    days = min(days, _MAX_DAYS)

    store = _get_lattice_store()
    activity = daily_write_counts(store, days=days)
    return jsonify({
        "days": activity.days,
        "buckets": [asdict(b) for b in activity.buckets],
        "total_nodes": total_node_count(store),
    }), 200
```

```javascript
  // ─── History grouping ─────────────────────────────────────────────────────
  const HISTORY_BUCKETS = ["Today", "Yesterday", "Previous 7 days", "Previous 30 days", "Older"];

  function bucketFor(updatedAt) {
    const now = new Date();
    const then = new Date(updatedAt);
    const days = Math.floor((now - then) / (1000 * 60 * 60 * 24));
    if (days < 1) return "Today";
    if (days < 2) return "Yesterday";
    if (days < 7) return "Previous 7 days";
    if (days < 30) return "Previous 30 days";
    return "Older";
  }

  async function loadSessions() {
    const r = await fetch(`/sessions?agent=${state.agent}&limit=100`);
    if (!r.ok) return;
    const data = await r.json();
    state.sessions = data.sessions || [];
    renderHistory();
  }

  function renderHistory() {
    const grouped = {};
    for (const b of HISTORY_BUCKETS) grouped[b] = [];
    const q = (document.querySelector('[data-testid="search"]').value || "").toLowerCase();
    for (const s of state.sessions) {
      if (q && !(s.title || "").toLowerCase().includes(q)) continue;
      grouped[bucketFor(s.updated_at)].push(s);
    }
    const host = elHistory();
    host.innerHTML = "";
    for (const bucket of HISTORY_BUCKETS) {
      const items = grouped[bucket];
      if (!items.length) continue;
      const h = document.createElement("h4");
      h.textContent = bucket;
      host.appendChild(h);
      for (const s of items) {
        const a = document.createElement("a");
        a.href = `/chat/${s.session_id}`;
        if (s.session_id === state.sessionId) a.className = "active";
        a.innerHTML = `<span class="dot" style="background:${AGENT_COLORS[s.agent] || "var(--gold)"}"></span>${s.title || "(untitled)"}`;
        host.appendChild(a);
      }
    }
  }
```
